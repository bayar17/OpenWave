cmake_minimum_required(VERSION 3.21)
project(OpenWaveWPEQtAdapter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.10 REQUIRED COMPONENTS Quick QuickPrivate GuiPrivate)
if(NOT OPENWAVE_SOURCE_ROOT)
    get_filename_component(OPENWAVE_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
endif()
set(W "${OPENWAVE_SOURCE_ROOT}/OceanLauncher/OceanBrowser/WPEWebKit")
set(A "${W}/Source/WebKit/UIProcess/API/wpe/qt6")
add_library(qtwpe SHARED "${A}/WPEDisplayQtQuick.cpp" "${A}/WPEToplevelQtQuick.cpp" "${A}/WPEViewQtQuick.cpp" "${A}/WPEQmlExtensionPlugin.cpp" "${A}/WPEQtView.cpp" "${A}/WPEQtViewLoadRequest.cpp")
target_compile_definitions(qtwpe PRIVATE QT_NO_KEYWORDS=1 QT_WPE_LIBRARY NDEBUG=1)
target_include_directories(qtwpe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${A}" "${W}/Source/WTF" "${W}/Source/bmalloc" "${W}/Source/WebKit/WPEPlatform" "${CMAKE_SYSROOT}/usr/include/wpe-webkit-2.0" "${CMAKE_SYSROOT}/usr/include/wpe-webkit-2.0/wpe-platform" "${CMAKE_SYSROOT}/usr/include/glib-2.0" "${CMAKE_SYSROOT}/usr/lib64/glib-2.0/include" "${CMAKE_SYSROOT}/usr/include/libsoup-3.0")
target_link_libraries(qtwpe PRIVATE Qt6::Quick Qt6::QuickPrivate Qt6::GuiPrivate epoxy gio-2.0 gobject-2.0 glib-2.0 WPEWebKit-2.0)
target_link_options(qtwpe PRIVATE -Wl,-z,defs)
install(TARGETS qtwpe LIBRARY DESTINATION lib64/qt6/qml/org/wpewebkit/qtwpe)
install(FILES "${A}/qmldir" DESTINATION lib64/qt6/qml/org/wpewebkit/qtwpe)
