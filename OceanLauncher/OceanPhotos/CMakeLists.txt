cmake_minimum_required(VERSION 3.21)
project(OceanPhotos VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.10 REQUIRED COMPONENTS Quick QuickControls2 QuickDialogs2 Svg)
qt_standard_project_setup(REQUIRES 6.10)
qt_add_executable(appOceanPhotos main.cpp GalleryLibrary.cpp GalleryLibrary.h)
qt_add_qml_module(appOceanPhotos
    URI OceanPhotos VERSION 1.0
    QML_FILES Main.qml GalleryGlyph.qml PhotoTile.qml PhotoPager.qml CrossfadeTitle.qml
    RESOURCES assets/icon.svg)
if(NOT CMAKE_CROSSCOMPILING)
set(font_files)
foreach(weight IN ITEMS Regular Light ExtraLight)
    set(font "${CMAKE_CURRENT_SOURCE_DIR}/../../userspace/fonts/assets/SourceSansPro-${weight}.otf")
    set_source_files_properties("${font}" PROPERTIES QT_RESOURCE_ALIAS "fonts/SourceSansPro-${weight}.otf")
    list(APPEND font_files "${font}")
endforeach()
qt_add_resources(appOceanPhotos gallery_fonts PREFIX /Gallery FILES ${font_files})
endif()
target_link_libraries(appOceanPhotos PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::QuickDialogs2 Qt6::Svg)
if(CMAKE_CROSSCOMPILING)
    find_package(OpenWaveSDK REQUIRED COMPONENTS Fonts FileChooser Applications Wallpaper)
    target_link_libraries(appOceanPhotos PRIVATE OpenWave::Fonts OpenWave::FileChooser OpenWave::Applications OpenWave::Wallpaper)
    target_compile_definitions(appOceanPhotos PRIVATE OPENWAVE_PHOTOS_SYSTEM_FONTS)
endif()
if(CMAKE_CROSSCOMPILING)
    include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/OpenWaveBundle.cmake")
    openwave_install_bundle(appOceanPhotos com.openwave.photos SYSTEM)
    install(FILES assets/icon.svg DESTINATION /System/Applications/com.openwave.photos/resources)
    install(FILES com.openwave.photos.desktop DESTINATION /usr/share/applications)
    install(FILES assets/icon.svg DESTINATION /usr/share/pixmaps RENAME com.openwave.photos.svg)
endif()
set_target_properties(appOceanPhotos PROPERTIES OUTPUT_NAME ocean-photos
    MACOSX_BUNDLE_GUI_IDENTIFIER com.openwave.photos MACOSX_BUNDLE TRUE WIN32_EXECUTABLE TRUE)
include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(gallery-library-tests tests/tst_library.cpp GalleryLibrary.cpp GalleryLibrary.h)
    target_link_libraries(gallery-library-tests PRIVATE Qt6::Gui Qt6::Test)
    add_test(NAME gallery-library COMMAND gallery-library-tests)
endif()
