cmake_minimum_required(VERSION 3.21)
project(PermissionTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Test Network Quick)
find_package(OpenWaveSDK REQUIRED COMPONENTS ApplicationServices)
add_executable(permission-policy-tests policy.cpp ../PolicyStore.cpp ../PermissionRegistry.cpp)
target_link_libraries(permission-policy-tests PRIVATE Qt6::Core Qt6::Test)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PipeWire REQUIRED IMPORTED_TARGET libpipewire-0.3)
add_executable(permission-runtime-tests runtime.cpp audio.c)
target_link_libraries(permission-runtime-tests PRIVATE PkgConfig::PipeWire)
target_link_libraries(permission-runtime-tests PRIVATE Qt6::Core Qt6::Quick OpenWave::ApplicationServices OpenWave::Identity)
foreach(target permission-policy-tests permission-runtime-tests)
  set_target_properties(${target} PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "/usr/lib64;/usr/lib")
endforeach()

find_package(LibXml2 REQUIRED)
foreach(service IN ITEMS promptd openwave-portal)
    add_executable(permission-${service}-test ../../${service}/main.c ../../xml/OpenWaveXml.c)
    target_link_libraries(permission-${service}-test PRIVATE OpenWave::Identity LibXml2::LibXml2)
    target_compile_definitions(permission-${service}-test PRIVATE
        OPENWAVE_DBUS_INTERFACE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../dbus/interfaces")
    set_target_properties(permission-${service}-test PROPERTIES AUTOMOC OFF)
endforeach()
target_compile_definitions(permission-promptd-test PRIVATE PROMPTD_TESTING)
target_compile_definitions(permission-openwave-portal-test PRIVATE PORTAL_TESTING)
